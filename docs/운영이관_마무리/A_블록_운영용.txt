  // ---------- 회원가입 (2단계) ----------
  // 예전에는 한 화면에서 이메일·비밀번호만 받고 바로 계정이 만들어졌다.
  // 가입 시점 동의(이용약관·개인정보·만14세)를 하나도 받지 않아 법적으로도 비어 있었다.
  function signupStepBar(){
    var dot = function(n, label){
      var on = signupStep === n, done = signupStep > n;
      var bg = (on || done) ? 'var(--accent)' : 'var(--border)';
      var fg = (on || done) ? '#fff' : 'var(--text2)';
      return '<span style="display:flex; align-items:center; gap:6px;">' +
          '<span aria-hidden="true" style="width:20px; height:20px; border-radius:50%; background:' + bg + '; color:' + fg +
            '; font-size:11px; font-weight:700; display:flex; align-items:center; justify-content:center;">' + (done ? '✓' : n) + '</span>' +
          '<span style="font-size:12px; font-weight:' + (on ? '700' : '400') + '; color:' + (on ? 'var(--text)' : 'var(--text2)') + ';">' + label + '</span>' +
        '</span>';
    };
    return '<div class="row" role="group" aria-label="가입 단계" style="align-items:center; gap:8px; margin:2px 0 14px;">' +
        dot(1, '계정 정보') +
        '<span aria-hidden="true" style="flex:1; height:1px; background:var(--border);"></span>' +
        dot(2, '약관 동의') +
      '</div>';
  }

  function signupHtml(){
    if(signupDone) return signupDoneHtml();
    return signupStep === 2 ? signupStep2Html() : signupStep1Html();
  }

  function pwRulesHtml(){
    var pw = authDraft.password || '';
    var rule = function(ok, text){
      return '<span style="font-size:11.5px; color:' + (ok ? 'var(--accent)' : 'var(--text2)') + ';">' +
             (ok ? '✓' : '·') + ' ' + text + '</span>';
    };
    return rule(pw.length >= 8, '8자 이상') + rule(/[A-Za-z]/.test(pw), '영문') + rule(/[0-9]/.test(pw), '숫자');
  }

  // 글자를 칠 때마다 폼을 다시 그리면 포커스가 날아간다. 바뀌는 조각만 손본다.
  function updateSignupStep1Ui(){
    var rules = document.getElementById('pw-rules');
    if(rules) rules.innerHTML = pwRulesHtml();
    var mm = document.getElementById('pw-mismatch');
    if(mm) mm.style.display = (authDraft.password2 && authDraft.password !== authDraft.password2) ? '' : 'none';
    syncAuthButtons();
  }

  // 가입 시점 동의는 계정이 생긴 뒤에야 서버에 남길 수 있다(원장이 owner_id 를 요구).
  // 인증 메일을 거치는 경우 세션이 바로 안 생기므로, 기기에 잠시 들고 있다가
  // 처음 로그인될 때 올린다.
  var PENDING_CONSENT_KEY = 'dadamom.pendingConsents';

  function stashSignupConsents(){
    var items = [
      { kind:'signup_terms',     granted:true },
      { kind:'signup_privacy',   granted:true },
      { kind:'signup_age14',     granted:true },
      { kind:'signup_marketing', granted: !!authDraft.marketing }
    ];
    try{ localStorage.setItem(PENDING_CONSENT_KEY, JSON.stringify(items)); }catch(e){}
    return items;
  }

  async function flushPendingConsents(){
    if(!sb || !authUser) return;
    var raw = null;
    try{ raw = localStorage.getItem(PENDING_CONSENT_KEY); }catch(e){ return; }
    if(!raw) return;
    var items;
    try{ items = JSON.parse(raw); }catch(e){ items = null; }
    if(!Array.isArray(items) || !items.length){
      try{ localStorage.removeItem(PENDING_CONSENT_KEY); }catch(e){}
      return;
    }
    await recordConsents(items);
    try{ localStorage.removeItem(PENDING_CONSENT_KEY); }catch(e){}
  }

  function handleSignupNext(){
    captureAuthDraft();
    if(!signupStep1Ok()) return;
    signupStep = 2;
    refreshAuthForms();
  }

  function handleSignupBack(){
    captureAuthDraft();
    signupStep = 1;
    refreshAuthForms();
  }

  async function handleSignupSubmit(){
    captureAuthDraft();
    var hint = document.getElementById('auth-hint');
    var btn = document.getElementById('signup-submit');
    if(!(authDraft.terms && authDraft.privacy && authDraft.age14)) return;
    if(!sb || !sb.auth || typeof sb.auth.signUp !== 'function'){
      if(hint) hint.textContent = '지금은 가입 기능을 사용할 수 없어요.';
      return;
    }
    if(btn){ btn.disabled = true; btn.textContent = '가입하는 중...'; }
    var email = (authDraft.email || '').trim();
    var nickname = (authDraft.nickname || '').trim();
    stashSignupConsents();
    try{
      var res = await sb.auth.signUp({
        email: email,
        password: authDraft.password,
        options: {
          emailRedirectTo: location.href.split('#')[0],
          data: nickname ? { nickname: nickname } : {}
        }
      });
      if(res.error){
        if(hint) hint.textContent = '회원가입 실패: ' + res.error.message;
        if(btn){ btn.disabled = false; btn.textContent = '가입 완료'; }
        try{ localStorage.removeItem(PENDING_CONSENT_KEY); }catch(e){}
        return;
      }
      var hasSession = !!(res.data && res.data.session);
      signupDone = { email: email, needVerify: !hasSession };
      refreshAuthForms();
      if(hasSession){
        await flushPendingConsents();
        closeModal();
      }
    }catch(e){
      if(hint) hint.textContent = '회원가입 중 문제가 생겼어요. 잠시 후 다시 시도해주세요.';
      if(btn){ btn.disabled = false; btn.textContent = '가입 완료'; }
    }
  }
